private pure function soft_w(work, id)

Linear weight of the active side of soft constraint id, i.e. what the multiplier has to exceed for the slack to become nonzero (a zero individual weight selects settings%w_soft).

    pure real(wp) function soft_w(work,id)

    type(daqp_workspace), intent(in) :: work !! workspace
    integer(ip), intent(in) :: id            !! constraint

    real(wp) :: w

    if (work%has_weights) then
        if (has(work%sense(id),daqp_lower)) then
            w = work%w_ls(id)
        else
            w = work%w_us(id)
        end if
        if (w /= 0.0_wp) then
            soft_w = w/work%scaling(id)
            return
        end if
    end if
    soft_w = work%settings%w_soft

    end function soft_w
