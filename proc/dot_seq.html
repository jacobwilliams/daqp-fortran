    pure real(wp) function dot_seq(n,a,b)

    integer(ip), intent(in) :: n    !! length
    real(wp), intent(in) :: a(n)    !! first vector
    real(wp), intent(in) :: b(n)    !! second vector

    integer(ip) :: i

    dot_seq = 0.0_wp
    do i = 1, n
        dot_seq = dot_seq + a(i)*b(i)
    end do

    end function dot_seq
