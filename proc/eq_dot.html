    pure real(wp) function eq_dot(a, b, n)

    real(wp), intent(in) :: a(:)  !! first vector
    real(wp), intent(in) :: b(:)  !! second vector
    integer(ip), intent(in) :: n  !! length

    integer(ip) :: i

    eq_dot = 0.0_wp
    do i = 1, n
        eq_dot = eq_dot + a(i)*b(i)
    end do

    end function eq_dot
