    real(wp) function elapsed_time(work)

    type(daqp_workspace), intent(in) :: work !! workspace

    integer(int64) :: count, rate

    call system_clock(count, rate)
    elapsed_time = real(count-work%timer_start,wp)/real(rate,wp)
